# mediapipe pulls in opencv-contrib-python as a dependency, which is what
# both video metadata reading (app.py) and pose analysis (pose_analysis.py)
# use for cv2 — don't also install opencv-python-headless, two OpenCV
# distributions in one venv conflict.
mediapipe==0.10.35

# Ask AI coaching box — optional "AI Chat" mode (Quick Answers mode needs
# no packages at all; each visitor brings their own API key for whichever
# provider they pick, see README). Any one of these three can be missing
# without breaking the others — ai_coach.py checks availability per provider.
anthropic
openai
google-genai

# "Add from YouTube" — downloads a clip straight into the library instead
# of requiring a local file. No system ffmpeg needed: youtube_fetch.py
# restricts to pre-muxed formats so yt-dlp never has to merge streams.
yt-dlp

# Bundles a static ffmpeg binary, so nothing here needs a system ffmpeg
# install. Used by video_trim.py (cutting to an exact, re-encoded H.264/AAC
# MP4) and by pose_analysis.py (re-encoding the elbow-trail overlay video —
# OpenCV's VideoWriter can only reliably write browser/mobile-incompatible
# codecs like mp4v, so the raw overlay gets muxed through ffmpeg afterward).
imageio-ffmpeg

# Desktop app (desktop.py) — wraps app.py in a native window instead of a
# browser tab. Not needed if you only ever run `python app.py`.
pywebview
